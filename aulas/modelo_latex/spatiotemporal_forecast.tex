\begin{frame}{Building the time series: resolution}
    \begin{columns}[t]
        % ---------------- spatial resolution
        \column{0.48\textwidth}
        \centering
        \textbf{Spatial resolution} $(H, W)$

        \vspace{0.1cm}
        \begin{tikzpicture}
            \foreach \n/\xs/\lab in {4/0/{coarse ($4 \times 4$)}, 12/2.6/{fine ($12 \times 12$)}} {
                \pgfmathsetmacro{\d}{2/\n}
                \foreach \i in {1,...,\n} {
                    \foreach \j in {1,...,\n} {
                        \pgfmathsetmacro{\cx}{(\i-0.5)/\n}
                        \pgfmathsetmacro{\cy}{(\j-0.5)/\n}
                        \pgfmathtruncatemacro{\v}{min(100, 100*exp(-((\cx-0.62)^2+(\cy-0.45)^2)/0.03) + 45*exp(-((\cx-0.25)^2+(\cy-0.7)^2)/0.01))}
                        \fill[blue!\v!white] (\xs+\d*\i-\d,\d*\j-\d) rectangle (\xs+\d*\i,\d*\j);
                    }
                }
                \draw[gray!70, step=\d] (\xs,0) grid (\xs+2,2);
                \draw (\xs,0) rectangle (\xs+2,2);
                \node[below, font=\tiny] at (\xs+1,0) {\lab};
            }
        \end{tikzpicture}

        \raggedright
        \scriptsize
        \begin{itemize}
            \item Coarse grids lose small structures (the small blob).
            \item Memory grows with $H \times W$ (4$\times$ for half the cell size).
            \item With $n$ pooling layers, $H$ and $W$ must be multiples of $2^n$: crop or pad. \\
                {\tiny e.g.\ $72 \rightarrow 36 \rightarrow 18 \rightarrow 9$ ($n=3$)}
        \end{itemize}

        % ---------------- temporal resolution
        \column{0.48\textwidth}
        \centering
        \textbf{Temporal resolution} $(\Delta t)$

        \vspace{0.1cm}
        \begin{tikzpicture}[x=0.6cm, y=0.8cm,
            declare function={f(\x)=0.5*sin(deg(1.2*\x)) + 1.2*exp(-((\x-3.75)^2)/0.08);}]
            \draw[->, gray] (0,-0.7) -- (6.4,-0.7) node[right, black, font=\tiny] {$t$};
            \draw[gray!60] plot[domain=0:6, samples=150, smooth] (\x,{f(\x)});
            % fine sampling
            \foreach \k in {0,...,24}
                \fill[blue] ({0.25*\k},{f(0.25*\k)}) circle (1.3pt);
            % coarse sampling
            \draw[red, thick] plot[mark=*, mark size=1.8pt] coordinates {
                (0,{f(0)}) (1.5,{f(1.5)}) (3,{f(3)}) (4.5,{f(4.5)}) (6,{f(6)})};
            \node[font=\tiny, anchor=west] at (4.3,1.4) {event};
        \end{tikzpicture}

        {\tiny \textcolor{blue}{$\bullet$ small $\Delta t$} \quad \textcolor{red}{$\bullet$ large $\Delta t$: event missed}}

        \raggedright
        \scriptsize
        \begin{itemize}
            \item $\Delta t$ must be compatible with the event: at least 2 samples per shortest period of interest (Nyquist).
            \item Smaller $\Delta t$ $\rightarrow$ more steps for the same horizon (longer sequences).
            \item Resample: average (smooth) or instantaneous values.
        \end{itemize}
    \end{columns}

    \vfill
    {\footnotesize Result: one tensor $(T, C, H, W)$ for the whole period, ready to be sliced into samples.}
\end{frame}

\begin{frame}{Recurrent Neural Networks evolution}
    \begin{itemize}
        \item ConvLSTM (2015)
        \begin{itemize}
            \item Uses a Memory Cell ($C$) that stores information through time steps.
            \item 2D convolutions on each frame.
        \end{itemize}
        \item PredRNN (2017) {\tiny extends ConvLSTM}
        \begin{itemize}
            \item ST-LSTM: Adds a new memory tensor ($M$) that is passed through stacked RNN layers.
        \end{itemize}
        \item E3D-LSTM (2018) {\tiny extends PredRNN}
        \begin{itemize}
            \item 3D Convolutions instead of 2D.
            \item Attention mechanism on tensor $C$.
        \end{itemize}
        \item PredRNN v2 (2022) {\tiny extends PredRNN}
        \begin{itemize}
            \item Adds a decoupled loss to learn different features with $C$ and $M$.
        \end{itemize}
        \item \textbf{Future?: XLSTM (2024) there is no ConvXLSTM yet.}
    \end{itemize}

    \vfill
    {\footnotesize References can be found in OpenSTL GitHub page.}
\end{frame}

\section{Generative and foundation models (2026)}
% ============================================================

\begin{frame}{From frame regression to generative prediction}
    \begin{columns}[t]
        \column{0.48\textwidth}
        \footnotesize
        \textbf{Deterministic models} {\tiny (previous section)}
        \begin{itemize}
            \item Trained with MAE/RMSE $\rightarrow$ predict the \emph{average} future $\rightarrow$ blur.
            \item Still improving: SimVP v2 and IAM4VP lead several benchmarks; PredFormer (2026) uses only gated transformers.
        \end{itemize}

        \vspace{0.15cm}
        \textbf{The generative turn}
        \begin{itemize}
            \item Diffusion models: sharp forecasts and \alert{many} plausible futures (ensembles).
            \item Flow matching / rectified flow: straight path from noise to data, plain regression loss, far fewer sampling steps.
        \end{itemize}

        \column{0.48\textwidth}
        \footnotesize
        \textbf{Large video models as world simulators}
        \begin{itemize}
            \item Closed: Veo 3.1, Seedance 2.0, Genie 3.
            \item Open: Wan 2.2, LTX-2.5, NVIDIA Cosmos, V-JEPA 2 (predicts in embedding space).
            \item Evaluated as future-state predictors (WorldReasonBench, 2026).
        \end{itemize}

        \vspace{0.15cm}
        \begin{block}{The shift}
            \footnotesize From ``predict the next frame'' to \textbf{``sample possible futures''}.
        \end{block}
    \end{columns}

    \vfill
    \hfill{\tiny (Tang et al. 2026; Lipman et al. 2023; Assran et al. 2025)}
\end{frame}

\begin{frame}{The same shift in weather forecasting}
    \begin{columns}[t]
        \column{0.48\textwidth}
        \footnotesize
        \textbf{Nowcasting (0--6 h): radar and satellite as video}
        \begin{itemize}
            \item Main recipe: \textbf{deterministic backbone + generative residual} (DiffCast, CasCast, CorrDiff).
            \item Flow matching for speed: FlowCast (ICLR 2026), FREUD (CVPR 2026).
            \item Satellite brightness temperature: diffusion on SEVIRI and GOES infrared images.
        \end{itemize}

        \column{0.48\textwidth}
        \footnotesize
        \textbf{Global forecasts (up to 15 days)}
        \begin{itemize}
            \item GenCast (diffusion, 50 members) beat ECMWF ENS on more than 96\% of targets.
            \item WeatherNext 3 (Aug 2026): 64 members, hourly updates.
            \item WeatherNext Cyclones: track, intensity and size.
        \end{itemize}

        \vspace{0.15cm}
        \textbf{In practice}
        \begin{itemize}
            \item Ensembles by default.
            \item Metrics beyond RMSE: CRPS, CSI, power spectra.
            \item Sampling time is the bottleneck.
        \end{itemize}
    \end{columns}

    \vfill
    \hfill{\tiny (Yu et al. 2024; Price et al. 2025; Chase et al. 2026; Afzali Gorooh et al. 2026; Alet et al. 2026)}
\end{frame}

\begin{frame}{How embeddings change the landscape}
    \begin{itemize}
        \item Embeddings become a \textbf{data layer}: download a vector per pixel instead of training an encoder.
        \item Good results with \textbf{few labels} (k-NN or a linear layer on top).

        \item Surface knowledge (terrain, land cover, water, cities) becomes a ready-made input for weather models.

\begin{frame}{Integrating embeddings in a forecast model}
    \small
    \begin{itemize}
        \item \textbf{A. Static conditioning}
        \begin{itemize}
            \item Pool the embeddings to the forecast grid (e.g.\ 10\,m $\rightarrow$ 2\,km), reduce with PCA, add as extra input channels (or FiLM).
            \item Cheapest option: works with any model from the previous section.
        \end{itemize}
        \item \textbf{B. Pretrained encoder}
        \begin{itemize}
            \item Replace the encoder with an Earth observation foundation model (TerraTorch, timm).
        \end{itemize}
        \item \textbf{C. Dynamic context}
        \begin{itemize}
            \item Forecasts or latents from Aurora / Prithvi WxC as conditioning, so the model knows the large-scale weather.
        \end{itemize}
        \item \textbf{D. Latent generation}
        \begin{itemize}
            \item Run diffusion / flow matching in a pretrained latent space to reduce cost.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Loss functions and evaluation metrics}
    \begin{columns}[t]
        \column{0.58\textwidth}
        \textbf{Numerical prediction (Regression)}
        \footnotesize
        \begin{align*}
            MAE &= \frac{\sum_{i=1}^{n} |y_i - \hat{y}_i|}{n} \\
            RMSE &= \sqrt{\frac{\sum_{i=1}^{n} (y_i - \hat{y}_i)^2}{n}} \\
            Huber &= \frac{\sum_{i=1}^{n} L_{huber}(y, \hat{y})}{n}
        \end{align*}
        where $L_{huber}(y,\hat{y}) =
        \begin{cases}
            \frac{1}{2}(y - \hat{y})^2 & \text{for } |y - \hat{y}| < \delta \\
            \delta \left(|y - \hat{y}| - \frac{1}{2}\delta\right) & \text{otherwise}
        \end{cases}$
        \[
        R^2 = 1 - \frac{\sum_{i=1}^{n} (y_i - \hat{y}_i)^2}{\sum_{i=1}^{n} (y_i - \bar{y})^2}
        \]

        \column{0.4\textwidth}
        \textbf{Categorical prediction}
        \begin{itemize}
            \item Categorical cross entropy loss
            \item Focal loss
            \item Precision, recall, F-score
            \item Intersection over union (IOU)
        \end{itemize}

        \vfill
        \scriptsize
        $n$: Number of samples in a batch\\
        $y$: Values from the training sample;\\
        $\hat{y}$: Values predicted with the network.
    \end{columns}
\end{frame}
