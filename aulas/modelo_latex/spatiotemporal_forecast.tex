\begin{frame}{Building the time series: resolution}
    \begin{columns}[t]
        % ---------------- spatial resolution
        \column{0.48\textwidth}
        \centering
        \textbf{Spatial resolution} $(H, W)$

        \vspace{0.1cm}
        \begin{tikzpicture}
            \foreach \n/\xs/\lab in {4/0/{coarse ($4 \times 4$)}, 12/2.6/{fine ($12 \times 12$)}} {
                \pgfmathsetmacro{\d}{2/\n}
                \foreach \i in {1,...,\n} {
                    \foreach \j in {1,...,\n} {
                        \pgfmathsetmacro{\cx}{(\i-0.5)/\n}
                        \pgfmathsetmacro{\cy}{(\j-0.5)/\n}
                        \pgfmathtruncatemacro{\v}{min(100, 100*exp(-((\cx-0.62)^2+(\cy-0.45)^2)/0.03) + 45*exp(-((\cx-0.25)^2+(\cy-0.7)^2)/0.01))}
                        \fill[blue!\v!white] (\xs+\d*\i-\d,\d*\j-\d) rectangle (\xs+\d*\i,\d*\j);
                    }
                }
                \draw[gray!70, step=\d] (\xs,0) grid (\xs+2,2);
                \draw (\xs,0) rectangle (\xs+2,2);
                \node[below, font=\tiny] at (\xs+1,0) {\lab};
            }
        \end{tikzpicture}

        \raggedright
        \scriptsize
        \begin{itemize}
            \item Coarse grids lose small structures (the small blob).
            \item Memory grows with $H \times W$ (4$\times$ for half the cell size).
            \item With $n$ pooling layers, $H$ and $W$ must be multiples of $2^n$: crop or pad. \\
                {\tiny e.g.\ $72 \rightarrow 36 \rightarrow 18 \rightarrow 9$ ($n=3$)}
        \end{itemize}

        % ---------------- temporal resolution
        \column{0.48\textwidth}
        \centering
        \textbf{Temporal resolution} $(\Delta t)$

        \vspace{0.1cm}
        \begin{tikzpicture}[x=0.6cm, y=0.8cm,
            declare function={f(\x)=0.5*sin(deg(1.2*\x)) + 1.2*exp(-((\x-3.75)^2)/0.08);}]
            \draw[->, gray] (0,-0.7) -- (6.4,-0.7) node[right, black, font=\tiny] {$t$};
            \draw[gray!60] plot[domain=0:6, samples=150, smooth] (\x,{f(\x)});
            % fine sampling
            \foreach \k in {0,...,24}
                \fill[blue] ({0.25*\k},{f(0.25*\k)}) circle (1.3pt);
            % coarse sampling
            \draw[red, thick] plot[mark=*, mark size=1.8pt] coordinates {
                (0,{f(0)}) (1.5,{f(1.5)}) (3,{f(3)}) (4.5,{f(4.5)}) (6,{f(6)})};
            \node[font=\tiny, anchor=west] at (4.3,1.4) {event};
        \end{tikzpicture}

        {\tiny \textcolor{blue}{$\bullet$ small $\Delta t$} \quad \textcolor{red}{$\bullet$ large $\Delta t$: event missed}}

        \raggedright
        \scriptsize
        \begin{itemize}
            \item $\Delta t$ must be compatible with the event: at least 2 samples per shortest period of interest (Nyquist).
            \item Smaller $\Delta t$ $\rightarrow$ more steps for the same horizon (longer sequences).
            \item Resample: average (smooth) or instantaneous values.
        \end{itemize}
    \end{columns}

    \vfill
    {\footnotesize Result: one tensor $(T, C, H, W)$ for the whole period, ready to be sliced into samples.}
\end{frame}

                \draw[blue] (\k,0) -- (\k,{acf(\k)});

            }
            \foreach \k in {0,24,48,72} \node[below, font=\tiny] at (\k,-0.33) {\k};
            \node[font=\tiny] at (36,-0.62) {lag $k$ (hours)};
            \node[font=\tiny, red, anchor=south] at (24,{acf(24)+0.03}) {24 h};
            \node[font=\tiny, red, anchor=south] at (48,{acf(48)+0.03}) {48 h};
        \end{tikzpicture}

        {\tiny Illustrative ACF of an hourly VTEC pixel.\\ Shaded: 95\% band of white noise.\par}

        \column{0.46\textwidth}
        \scriptsize
        \[
            \rho(k) = \operatorname{corr}(x_t,\, x_{t+k})
        \]
        \begin{itemize}
            \item \textbf{Peaks} $\rightarrow$ seasonal period (24 h): $t$ should cover at least one cycle.
            \item \textbf{Decay} $\rightarrow$ how far ahead the past is informative ($t'$).
            \item \textbf{PACF}: direct effect of lag $k$, removing intermediate lags.
            \item \textbf{For maps}: ACF of each pixel, or $\operatorname{corr}(X_t, X_{t+k})$ over all pixels $=$ skill of the \emph{persistence} baseline $\hat{X}_{t+k} = X_t$.
        \end{itemize}
    \end{columns}

\begin{frame}{Diffusion video prediction: core idea}
    \centering
    \begin{tikzpicture}[font=\scriptsize]
        \pgfmathsetseed{7}
        % five versions of the future clip, from data (a=0) to pure noise (a=1)
        \foreach \k/\a in {0/0, 1/0.3, 2/0.55, 3/0.8, 4/1} {
            \begin{scope}[shift={(2.1*\k,0)}]
                \foreach \i in {0,...,7} {
                    \foreach \j in {0,...,5} {
                        \pgfmathsetmacro{\cx}{(\i+0.5)/8}
                        \pgfmathsetmacro{\cy}{(\j+0.5)/6}
                        \pgfmathsetmacro{\b}{exp(-((\cx-0.4)^2+(\cy-0.5)^2)/0.04)}
                        \pgfmathtruncatemacro{\v}{100*((1-\a)*\b + \a*rnd)}
                        \fill[blue!\v!white] (0.18*\i,0.18*\j) rectangle (0.18*\i+0.18,0.18*\j+0.18);
                    }
                }
                \draw (0,0) rectangle (1.44,1.08);
            \end{scope}
        }
        \node[below] at (0.72,0) {$Y_0$: future frames};
        \node[below] at (4.92,0) {$Y_\tau$};
        \node[below] at (9.12,0) {$Y_T \sim \mathcal{N}(0, I)$};
        % forward and reverse processes
        \draw[-{Stealth}, thick, gray] (0.72,1.3) to[bend left=12] node[above, black] {forward: add noise (fixed, no learning)} (9.12,1.3);
        \draw[-{Stealth}, thick, blue!70!black] (9.12,-0.45) to[bend left=12] node[below, black] {reverse: denoise step by step with a network $\epsilon_\theta(Y_\tau, \tau, X)$} (0.72,-0.45);
    \end{tikzpicture}

    \scriptsize
    \begin{columns}[t]
        \column{0.48\textwidth}
        \textbf{Training}
        \begin{itemize}
            \item Sample a step $\tau$ and noise $\epsilon \sim \mathcal{N}(0, I)$.
            \item Noisy target: $Y_\tau = \sqrt{\bar\alpha_\tau}\, Y + \sqrt{1-\bar\alpha_\tau}\, \epsilon$
            \item Predict the noise, given the past $X$: \\
                $\mathcal{L} = \lVert \epsilon - \epsilon_\theta(Y_\tau, \tau, X) \rVert^2$
        \end{itemize}

        \column{0.48\textwidth}
        \textbf{Sampling (forecast)}
        \begin{itemize}
            \item Start from noise $Y_T$, remove a little of the predicted noise at each step.
            \item $T \approx 1000$ steps (DDPM) or 20--50 (DDIM).
            \item New noise $\rightarrow$ a new plausible future: \alert{ensembles for free}.
        \end{itemize}
    \end{columns}

    \vfill
    {\tiny $\tau$ is the \emph{diffusion step}, not the forecast time $t$. \hfill (Ho et al. 2020; Song et al. 2021)}
\end{frame}

\begin{frame}{Flow matching video prediction: core idea}
    \begin{columns}[c]
        \column{0.44\textwidth}
        \centering
        \begin{tikzpicture}[font=\scriptsize, scale=0.8]
            \fill[gray!20] (0,0) circle (0.9);
            \fill[blue!20] (4,0.2) ellipse (0.6 and 1.1);
            \node at (0,-1.15) {noise $Z$, $\tau=0$};
            \node at (4,-1.15) {data $Y$, $\tau=1$};
            \foreach \zx/\zy/\yx/\yy in {-0.4/0.5/3.8/1.0, 0.3/0.3/4.2/0.6, -0.2/-0.3/3.9/-0.1, 0.4/-0.5/4.3/-0.6} {
                \fill[gray!70!black] (\zx,\zy) circle (1.5pt);
                \fill[blue] (\yx,\yy) circle (1.5pt);
                \draw[-{Stealth}, blue!70!black] (\zx,\zy) -- (\yx,\yy);
            }
            \draw[-{Stealth}, red, dashed] (0.1,0.7) .. controls (1.2,2.1) and (2.4,-1.6) .. (3.85,0.3);
            \node[red, font=\tiny] at (2.0,1.5) {diffusion: curved path};
            \node[blue!70!black, font=\tiny] at (2.0,-1.6) {flow: straight paths};
        \end{tikzpicture}

        \column{0.54\textwidth}
        \scriptsize
        \begin{itemize}
            \item \textbf{Path}: a straight line between a noise sample and a future clip
            \[ Y_\tau = (1-\tau)\, Z + \tau\, Y, \quad \tau \in [0, 1] \]
            \item \textbf{Velocity} along the path is constant: $dY_\tau / d\tau = Y - Z$
            \item \textbf{Training}: plain regression of the velocity
            \[ \mathcal{L} = \lVert v_\theta(Y_\tau, \tau, X) - (Y - Z) \rVert^2 \]
            \item \textbf{Sampling}: solve the ODE from $\tau=0$ to $1$ with Euler steps
            \[ Y \leftarrow Y + \Delta\tau \; v_\theta(Y, \tau, X) \]
        \end{itemize}
    \end{columns}

    \begin{block}{Why it matters for forecasting}
        \scriptsize Straight paths are easy to follow: \textbf{1--20 steps} instead of hundreds. Different $Z$ $\rightarrow$ different futures (ensemble). \hfill{\tiny (Lipman et al. 2023; Liu et al. 2023)}
    \end{block}
\end{frame}

\begin{frame}[fragile]{Diffusion vs.\ flow matching in code}
    \begin{columns}[t]
        \column{0.49\textwidth}
        \footnotesize\textbf{Diffusion (predict the noise)}
\begin{minted}[fontsize=\tiny, bgcolor=codegray]{python}
# X: past (B,t,C,H,W)   Y: future (B,t',C,H,W)
tau = torch.randint(0, T, (B,))
eps = torch.randn_like(Y)
a = alpha_bar[tau].view(B, 1, 1, 1, 1)
Y_tau = a.sqrt() * Y + (1 - a).sqrt() * eps
loss = mse(model(Y_tau, tau, X), eps)
\end{minted}

        \column{0.49\textwidth}
        \footnotesize\textbf{Flow matching (predict the velocity)}
\begin{minted}[fontsize=\tiny, bgcolor=codegray]{python}
# X: past (B,t,C,H,W)   Y: future (B,t',C,H,W)
tau = torch.rand(B).view(B, 1, 1, 1, 1)
Z = torch.randn_like(Y)
Y_tau = (1 - tau) * Z + tau * Y
loss = mse(model(Y_tau, tau, X), Y - Z)
\end{minted}
        \footnotesize\textbf{Forecast with $n$ Euler steps}
\begin{minted}[fontsize=\tiny, bgcolor=codegray]{python}
Y = torch.randn(B, t2, C, H, W)      # noise
for k in range(n):
    Y = Y + model(Y, k / n, X) / n   # follow v
\end{minted}
    \end{columns}

    \vspace{0.15cm}
    \centering
    \scriptsize
    \begin{tabular}{@{}lll@{}}
        \toprule
        & \textbf{Diffusion} & \textbf{Flow matching} \\
        \midrule
        Network predicts & noise $\epsilon$ & velocity $Y - Z$ \\
        Path noise $\rightarrow$ data & curved (noise schedule $\bar\alpha_\tau$) & straight line \\

        \bottomrule
    \end{tabular}

\begin{frame}{How embeddings change the landscape}
    \begin{itemize}
        \item Embeddings become a \textbf{data layer}: download a vector per pixel instead of training an encoder.
        \item Good results with \textbf{few labels} (k-NN or a linear layer on top).
        \item New models arrived roughly every two weeks in early 2026.
        \item Surface knowledge (terrain, land cover, water, cities) becomes a ready-made input for weather models.
    \end{itemize}

\begin{frame}{Integrating embeddings in a forecast model}
    \small
    \begin{itemize}
        \item \textbf{A. Static conditioning}
        \begin{itemize}
            \item Pool the embeddings to the forecast grid (e.g.\ 10\,m $\rightarrow$ 2\,km), reduce with PCA, add as extra input channels (or FiLM).
            \item Cheapest option: works with any model from the previous section.
        \end{itemize}
        \item \textbf{B. Pretrained encoder}
        \begin{itemize}
            \item Replace the encoder with an Earth observation foundation model (TerraTorch, timm).
        \end{itemize}
        \item \textbf{C. Dynamic context}
        \begin{itemize}
            \item Forecasts or latents from Aurora / Prithvi WxC as conditioning, so the model knows the large-scale weather.
        \end{itemize}
        \item \textbf{D. Latent generation}
        \begin{itemize}
            \item Run diffusion / flow matching in a pretrained latent space to reduce cost.
        \end{itemize}
    \end{itemize}
\end{frame}

        \toprule
         & Prithvi {\normalfont (NASA / IBM)} & Cosmos {\normalfont (NVIDIA)} \\
        \midrule
        Learns from & Physical Earth data: reanalysis (WxC) and satellite imagery (EO-2.0) & Huge amounts of everyday video \\
        Understands & Physical variables and real geography & Generic visual motion: objects, cameras, scenes \\
        Predicts & Atmospheric and surface fields & Future video frames (and robot actions in Cosmos 3) \\

    \end{tabularx}

    \espaco
    \begin{block}{Bottom line}
        \footnotesize Cosmos has the broader impact across AI; \textbf{Prithvi has the greater impact on Earth science} and is the better starting point for weather forecasting.
    \end{block}
\end{frame}

\begin{frame}{Summary before the hands on}
    \begin{enumerate}
        \item The field moved from \textbf{deterministic frame regression} to \textbf{generative sampling of possible futures}.
        \item Weather systems converged on one recipe: \textbf{deterministic backbone + generative residual}, evaluated with CRPS, CSI and spectra.
        \item \textbf{Earth embeddings} turn the land surface into a ready-made input layer.
        \item Main gaps: time-scale mismatch (annual vs.\ minutes), licensing, evaluation cost.
    \end{enumerate}

    \espaco
    \begin{block}{Next: hands on}
        \footnotesize The OpenSTL models are the \textbf{deterministic backbone}. A generative residual or embedding channels can be added on top of them.
    \end{block}
\end{frame}

\begin{frame}{Loss functions and evaluation metrics}
    \begin{columns}[t]
        \column{0.58\textwidth}
        \textbf{Numerical prediction (Regression)}
        \footnotesize
        \begin{align*}
            MAE &= \frac{\sum_{i=1}^{n} |y_i - \hat{y}_i|}{n} \\
            RMSE &= \sqrt{\frac{\sum_{i=1}^{n} (y_i - \hat{y}_i)^2}{n}} \\
            Huber &= \frac{\sum_{i=1}^{n} L_{huber}(y, \hat{y})}{n}
        \end{align*}
        where $L_{huber}(y,\hat{y}) =
        \begin{cases}
            \frac{1}{2}(y - \hat{y})^2 & \text{for } |y - \hat{y}| < \delta \\
            \delta \left(|y - \hat{y}| - \frac{1}{2}\delta\right) & \text{otherwise}
        \end{cases}$
        \[
        R^2 = 1 - \frac{\sum_{i=1}^{n} (y_i - \hat{y}_i)^2}{\sum_{i=1}^{n} (y_i - \bar{y})^2}
        \]

        \column{0.4\textwidth}
        \textbf{Categorical prediction}
        \begin{itemize}
            \item Categorical cross entropy loss
            \item Focal loss
            \item Precision, recall, F-score
            \item Intersection over union (IOU)
        \end{itemize}

        \vfill
        \scriptsize
        $n$: Number of samples in a batch\\
        $y$: Values from the training sample;\\
        $\hat{y}$: Values predicted with the network.
    \end{columns}
\end{frame}
